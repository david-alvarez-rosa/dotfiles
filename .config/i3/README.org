i3 is a tiling window manager, completely written from scratch. The target
platforms are GNU/Linux and BSD operating systems, the code is Free and Open
Source Software (FOSS) under the BSD license. i3 is primarily targeted at
advanced users and developers. This is my personal configuration, primarily
based on the principles of minimalism (simplicity is the key for success). All
key bindings are set following the holy Emacs philosophy.

* Theme
Set default font (for window title).
#+begin_src conf
  font pango:monospace 11
#+end_src

Set initial inner and outer gaps and window border
#+begin_src conf
  gaps inner 4
  gaps outer 2
  default_border pixel 1
#+end_src

Use a uniform dark border on every side.
#+begin_src conf
  #                       border  backgr. text    indicator child
  client.focused          #555555 #555555 #ffffff #555555   #555555
  client.focused_inactive #222222 #222222 #888888 #222222   #222222
  client.unfocused        #222222 #222222 #888888 #222222   #222222
  client.urgent           #900000 #900000 #ffffff #900000   #900000
#+end_src

* Functionality
Set modifier key to Super (Windows key).
#+begin_src conf
  set $mod Mod4
#+end_src

Disable focus wrapping (I have always found this feature annoying).
#+begin_src conf
  focus_wrapping no
#+end_src

Set workspace names.
#+begin_src conf
  set $ws1 "1. Emacs"

  set $ws7 "7. Emacs"
  set $ws8 "8. Browser"
  set $ws9 "9. Terminal"
  set $ws10 "10. Executing"
#+end_src

* Auto Start
Auto start Emacs on workspace 1 and Firefox on workspace 2.
#+begin_src conf
  exec emacs
  exec firefox
  exec --no-startup-id sh -c 'until i3-msg [class=firefox] move container to workspace $ws2; do sleep 0.5; done'
#+end_src

Start the Nextcloud client in the background, once KeePassXC has unlocked the
keychain it reads its token from (waits until the SSH key is loaded).
#+begin_src conf
  exec --no-startup-id sh -c 'until ssh-add -l >/dev/null 2>&1; do sleep 0.5; done; nextcloud --background'
#+end_src

Lock the screen with =i3lock= before sleep/suspend. =xss-lock= holds a sleep
inhibitor until the screen is locked.
#+begin_src conf
  exec --no-startup-id xss-lock --transfer-sleep-lock -- i3lock -i ~/media/wallpaper.png -u
#+end_src

Set the wallpaper.
#+begin_src conf
  exec_always --no-startup-id feh --no-fehbg --bg-fill ~/media/wallpaper.png
#+end_src

* Key bindings
** Apps
Open a terminal
#+begin_src conf
  bindsym $mod+Return exec emacsclient -c
  bindsym $mod+Shift+Return exec emacsclient -c -e "(ghostel t)"
#+end_src

Application launcher and manual screen lock.
#+begin_src conf
  bindsym $mod+r exec --no-startup-id rofi -show drun
  bindsym $mod+l exec --no-startup-id i3lock -i ~/media/wallpaper.png -u
#+end_src

Toggle between light and dark mode.
#+begin_src conf
  bindsym $mod+d exec --no-startup-id dark-mode-toggle
#+end_src

Toggle keyboard layout between US and Spanish.
#+begin_src conf
  bindsym $mod+space exec --no-startup-id "setxkbmap -query | grep -qE 'layout:[[:space:]]*es' && setxkbmap -layout us,es -option '' -option compose:ralt || setxkbmap -layout es,us -option '' -option compose:ralt"
#+end_src

Volume, microphone, brightness, touchpad, zoom and music keys.
#+begin_src conf
  bindsym XF86AudioRaiseVolume exec --no-startup-id "v=$(wpctl get-volume @DEFAULT_AUDIO_SINK@|awk '{print int($2*100)}');n=$(((v+12)/5*5));wpctl set-volume @DEFAULT_AUDIO_SINK@ $n%;wpctl set-mute @DEFAULT_AUDIO_SINK@ 0"
  bindsym Control+XF86AudioRaiseVolume exec --no-startup-id "v=$(wpctl get-volume @DEFAULT_AUDIO_SINK@|awk '{print int($2*100)}');n=$(((v+7)/5*5));wpctl set-volume @DEFAULT_AUDIO_SINK@ $n%;wpctl set-mute @DEFAULT_AUDIO_SINK@ 0"
  bindsym XF86AudioLowerVolume exec --no-startup-id "v=$(wpctl get-volume @DEFAULT_AUDIO_SINK@|awk '{print int($2*100)}');n=$(((v-8)/5*5));[ $n -lt 0 ]&&n=0;wpctl set-volume @DEFAULT_AUDIO_SINK@ $n%"
  bindsym Control+XF86AudioLowerVolume exec --no-startup-id "v=$(wpctl get-volume @DEFAULT_AUDIO_SINK@|awk '{print int($2*100)}');n=$(((v-3)/5*5));[ $n -lt 0 ]&&n=0;wpctl set-volume @DEFAULT_AUDIO_SINK@ $n%"
  bindsym XF86AudioMute exec --no-startup-id "wpctl set-volume @DEFAULT_AUDIO_SINK@ 0; wpctl set-mute @DEFAULT_AUDIO_SINK@ toggle"
  bindsym XF86AudioMicMute exec --no-startup-id "wpctl set-mute @DEFAULT_AUDIO_SOURCE@ toggle"
  bindsym XF86MonBrightnessUp exec --no-startup-id "v=$(brightnessctl -m|cut -d, -f4|tr -d %);if [ $v -lt 1 ];then n=1;elif [ $v -lt 10 ];then n=10;elif [ $v -lt 20 ];then n=20;else n=$((v/20*20+20));[ $n -gt 100 ]&&n=100;fi;brightnessctl -q set $n%"
  bindsym XF86MonBrightnessDown exec --no-startup-id "v=$(brightnessctl -m|cut -d, -f4|tr -d %);if [ $v -le 1 ];then n=0;elif [ $v -le 10 ];then n=1;elif [ $v -le 20 ];then n=10;else n=$(((v+19)/20*20-20));fi;brightnessctl -q set $n%"
  bindsym XF86Display exec --no-startup-id "id=$(xinput list | grep -i touchpad | grep -o 'id=[0-9]*' | head -1 | cut -d= -f2); xinput list-props $id | grep -q 'Device Enabled.*1$' && xinput disable $id || xinput enable $id"
  bindsym XF86Favorites exec --no-startup-id "if [ -e /tmp/.zoom ]; then xrandr --output eDP-1 --auto; rm /tmp/.zoom; else xrandr --output eDP-1 --mode 1280x800; touch /tmp/.zoom; fi; feh --no-fehbg --bg-fill ~/media/wallpaper.png"
  bindsym XF86AudioPlay exec --no-startup-id "mpc toggle"
  bindsym XF86AudioPrev exec --no-startup-id "mpc prev"
  bindsym XF86AudioNext exec --no-startup-id "mpc next"
#+end_src

Screenshots to =~/tmp= and the clipboard.
#+begin_src conf
  bindsym Print exec --no-startup-id "maim -u | tee ~/tmp/$(date +'%Y%m%d-%H:%M:%S').png | xclip -selection clipboard -t image/png"
  bindsym Shift+Print exec --no-startup-id "maim -u --window $(xdotool getactivewindow) | tee ~/tmp/$(date +'%Y%m%d-%H:%M:%S').png | xclip -selection clipboard -t image/png"
  bindsym --release Control+Print exec --no-startup-id "maim -u --select | tee ~/tmp/$(date +'%Y%m%d-%H:%M:%S').png | xclip -selection clipboard -t image/png"
#+end_src

** Windows
Key bindings related to windows (inside a workspace).

Killing current window.
#+begin_src conf
  bindsym $mod+q kill
#+end_src

*** Window focus
Set key bindings for changing window focus (following the holy Emacs "bfnp"
moving set).
#+begin_src conf
  bindsym $mod+b exec emacs-i3-integration focus left
  bindsym $mod+n exec emacs-i3-integration focus down
  bindsym $mod+p exec emacs-i3-integration focus up
  bindsym $mod+f exec emacs-i3-integration focus right
#+end_src

Change focus between tiling/floating windows.
#+begin_src conf
  bindsym $mod+t focus mode_toggle
#+end_src

*** Window moving
Set key bindings for moving windows around with same commands as window focus,
but with =Shift= modifier.
#+begin_src conf
  bindsym $mod+Shift+b exec emacs-i3-integration move left 20px
  bindsym $mod+Shift+n exec emacs-i3-integration move down 20px
  bindsym $mod+Shift+p exec emacs-i3-integration move up 20px
  bindsym $mod+Shift+f exec emacs-i3-integration move right 20px
#+end_src

Use $mod+Mouse for moving floating windows.
#+begin_src conf
  floating_modifier $mod
#+end_src

*** Window resize
Set key bindings for window resizing with same commands as window focus, but
with =Control= modifier.
  #+begin_src conf
  bindsym $mod+Control+f exec --no-startup-id "~/.local/bin/i3resize right"
  bindsym $mod+Control+p exec --no-startup-id "~/.local/bin/i3resize up"
  bindsym $mod+Control+n exec --no-startup-id "~/.local/bin/i3resize down"
  bindsym $mod+Control+b exec --no-startup-id "~/.local/bin/i3resize left"
  #+end_src

*** Window video multitasking
The following creates a small floating sticky window in the bottom right of the
screen. Very useful for multitasking while watching a video (not recommended
for productivity, though).
#+begin_src conf
  bindsym $mod+Shift+v floating toggle; [con_id=__focused__ floating] sticky enable, resize set 25 ppt 25 ppt, move position 75 ppt 75 ppt, move left 5 px, move up 5 px
#+end_src
