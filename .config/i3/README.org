* Auto Start
Emacs on workspace 1, Firefox on 2.
#+begin_src conf
  exec emacs
  exec firefox
  exec --no-startup-id sh -c 'until i3-msg [class=firefox] move container to workspace $ws2; do sleep 0.5; done'
#+end_src

Nextcloud waits for KeePassXC to unlock (SSH key loaded).
#+begin_src conf
  exec --no-startup-id sh -c 'until ssh-add -l >/dev/null 2>&1; do sleep 0.5; done; nextcloud --background'
#+end_src

Lock before suspend.
#+begin_src conf

#+end_src

#+begin_src conf

US/Spanish layout toggle.
#+begin_src conf
  bindsym $mod+space exec --no-startup-id "setxkbmap -query | grep -qE 'layout:[[:space:]]*es' && setxkbmap -layout us,es -option '' -option compose:ralt || setxkbmap -layout es,us -option '' -option compose:ralt; pkill -RTMIN+2 i3blocks"

Media, brightness, touchpad and zoom keys.
#+begin_src conf
  bindsym XF86AudioRaiseVolume exec --no-startup-id "v=$(wpctl get-volume @DEFAULT_AUDIO_SINK@|awk '{print int($2*100)}');n=$(((v+12)/5*5));wpctl set-volume @DEFAULT_AUDIO_SINK@ $n%;wpctl set-mute @DEFAULT_AUDIO_SINK@ 0;pkill -RTMIN+1 i3blocks"
  bindsym Control+XF86AudioRaiseVolume exec --no-startup-id "v=$(wpctl get-volume @DEFAULT_AUDIO_SINK@|awk '{print int($2*100)}');n=$(((v+7)/5*5));wpctl set-volume @DEFAULT_AUDIO_SINK@ $n%;wpctl set-mute @DEFAULT_AUDIO_SINK@ 0;pkill -RTMIN+1 i3blocks"
  bindsym XF86AudioLowerVolume exec --no-startup-id "v=$(wpctl get-volume @DEFAULT_AUDIO_SINK@|awk '{print int($2*100)}');n=$(((v-8)/5*5));[ $n -lt 0 ]&&n=0;wpctl set-volume @DEFAULT_AUDIO_SINK@ $n%;pkill -RTMIN+1 i3blocks"
  bindsym Control+XF86AudioLowerVolume exec --no-startup-id "v=$(wpctl get-volume @DEFAULT_AUDIO_SINK@|awk '{print int($2*100)}');n=$(((v-3)/5*5));[ $n -lt 0 ]&&n=0;wpctl set-volume @DEFAULT_AUDIO_SINK@ $n%;pkill -RTMIN+1 i3blocks"
  bindsym XF86AudioMute exec --no-startup-id "wpctl set-volume @DEFAULT_AUDIO_SINK@ 0; wpctl set-mute @DEFAULT_AUDIO_SINK@ toggle; pkill -RTMIN+1 i3blocks"
  bindsym XF86AudioMicMute exec --no-startup-id "wpctl set-mute @DEFAULT_AUDIO_SOURCE@ toggle"
  bindsym XF86MonBrightnessUp exec --no-startup-id "v=$(brightnessctl -m|cut -d, -f4|tr -d %);if [ $v -lt 1 ];then n=1;elif [ $v -lt 10 ];then n=10;elif [ $v -lt 20 ];then n=20;else n=$((v/20*20+20));[ $n -gt 100 ]&&n=100;fi;brightnessctl -q set $n%"
  bindsym XF86MonBrightnessDown exec --no-startup-id "v=$(brightnessctl -m|cut -d, -f4|tr -d %);if [ $v -le 1 ];then n=0;elif [ $v -le 10 ];then n=1;elif [ $v -le 20 ];then n=10;else n=$(((v+19)/20*20-20));fi;brightnessctl -q set $n%"
  bindsym XF86Display exec --no-startup-id "id=$(xinput list | grep -i touchpad | grep -o 'id=[0-9]*' | head -1 | cut -d= -f2); xinput list-props $id | grep -q 'Device Enabled.*1$' && xinput disable $id || xinput enable $id"
  bindsym XF86Favorites exec --no-startup-id zoom-toggle
  bindsym Shift+XF86Favorites exec --no-startup-id zoom-toggle prev
  bindsym XF86AudioPlay exec --no-startup-id "mpc toggle"
  bindsym XF86AudioPrev exec --no-startup-id "mpc prev"
  bindsym XF86AudioNext exec --no-startup-id "mpc next"
#+end_src

Screenshots to =~/tmp= and the clipboard.
#+begin_src conf
  bindsym Print exec --no-startup-id "maim -u | tee ~/tmp/$(date +'%Y%m%d-%H:%M:%S').png | xclip -selection clipboard -t image/png"
  bindsym Shift+Print exec --no-startup-id "maim -u --window $(xdotool getactivewindow) | tee ~/tmp/$(date +'%Y%m%d-%H:%M:%S').png | xclip -selection clipboard -t image/png"
  bindsym --release Control+Print exec --no-startup-id "maim -u --select | tee ~/tmp/$(date +'%Y%m%d-%H:%M:%S').png | xclip -selection clipboard -t image/png"
#+end_src
