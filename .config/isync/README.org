Two-way IMAP/Maildir sync for mu4e.  Needs this line in
=~/.local/share/authinfo.gpg=:
#+begin_src conf :tangle no
  machine mail.alvarezrosa.com login david@alvarezrosa.com port 465 password "***pass***"
#+end_src

After the first sync, initialize mu:
#+begin_src conf :tangle no
  mu init -m ~/.local/share/mail --my-address=david@alvarezrosa.com
#+end_src

* Personal
#+begin_src conf
  IMAPAccount Personal
  Host mail.alvarezrosa.com
  User david@alvarezrosa.com
  PassCmd "gpg --no-tty -qd ~/.local/share/authinfo.gpg | grep mail.alvarezrosa.com | awk '{print $8}'"
  TLSType IMAPS
  DisableExtension COMPRESS=DEFLATE

  IMAPStore personal-remote
  Account   Personal
  UseNamespace no
  PathDelimiter .

  MaildirStore personal-local
  Path ~/.local/share/mail/Personal/
  Inbox ~/.local/share/mail/Personal/Inbox
  SubFolders Verbatim

  Channel personal
  Far :personal-remote:
  Near :personal-local:
  Patterns *
  Create Both
  Expunge Both
  SyncState *
#+end_src
